<script>
/* CONFIG */
const MAX_PLAYERS = 10;
const PLAYER_COLORS = [
    "#00eaff","#ff0077","#ffcc00","#00ff66","#ff6600",
    "#aa00ff","#00ffaa","#ff0044","#66ff00","#0099ff"
];

/* NEON CATEGORY COLORS */
const categoryColors = {
    "Housing": "#00A8FF",
    "Cars": "#FF2E2E",
    "Avatar": "#FF00C8",
    "Life": "#00FF6A",
    "Job": "#FFE600",
    "Car Dealer": "#FF7A00",
    "Car Updates": "#B300FF",
    "House Updates": "#00FFD5",
    "Craigslist": "#9D00FF",
    "Doordash": "#FF5500",
    "Job Fare": "#FFB800"
};

/* ASK HOW MANY PLAYERS */
let playerCount = parseInt(prompt("How many players? (1-10)", "3"), 10);
if (isNaN(playerCount) || playerCount < 1) playerCount = 1;
if (playerCount > MAX_PLAYERS) playerCount = MAX_PLAYERS;

/* CATEGORIES */
const categories = Object.keys(categoryColors);

/* SAMPLE DECKS (replace with your real 440 if needed) */
let decks = {};
categories.forEach(cat => {
    decks[cat] = [];
    for (let i = 1; i <= 40; i++) {
        decks[cat].push({
            category: cat,
            type: i % 2 === 0 ? "GOOD" : "BAD",
            title: `${cat} Event ${i}`,
            effect: (i % 2 === 0 ? "+" : "-") + (10 + i)
        });
    }
});

/* BOARD DATA */
let boardSpaces = [];
const boardEl = document.getElementById("board");

/* PLAYERS DATA */
let players = [];
let currentPlayerIndex = 0;

/* INIT PLAYERS + MINI AVATARS + TOKENS */
function initPlayers() {
    const container = document.getElementById("boardContainer");
    const miniStrip = document.getElementById("miniAvatars");

    for (let i = 0; i < playerCount; i++) {

        /* ASK FOR PLAYER NAME */
        let name = prompt("Enter name for Player " + (i + 1), "Player " + (i + 1));

        players.push({
            id: i + 1,
            name: name || ("Player " + (i + 1)),
            pos: 0,
            money: 500,
            happiness: 5,
            job: "None",
            car: "None",
            house: "None",
            tokenEl: null,
            miniEl: null
        });

        /* BOARD TOKEN */
        let token = document.createElement("div");
        token.className = "playerToken";
        token.style.background = PLAYER_COLORS[i];
        container.appendChild(token);
        players[i].tokenEl = token;

        /* MINI AVATAR */
        let mini = document.createElement("div");
        mini.className = "miniAvatar";
        mini.style.background = PLAYER_COLORS[i];
        mini.innerText = players[i].name;
        miniStrip.appendChild(mini);
        players[i].miniEl = mini;
    }
}

/* BUILD BOARD (RANDOM EDGE SPACES) */
function buildBoard() {
    boardSpaces = [];
    boardEl.innerHTML = "";

    for (let i = 0; i < 144; i++) {
        let isEdge =
            i < 12 || i >= 132 ||
            i % 12 === 0 || i % 12 === 11;

        if (isEdge) {
            let cat = categories[Math.floor(Math.random() * categories.length)];
            boardSpaces.push(cat);

            let div = document.createElement("div");
            div.className = "space";
            div.innerText = cat;
            div.style.borderColor = categoryColors[cat];
            div.style.color = categoryColors[cat];
            boardEl.appendChild(div);
        } else {
            let div = document.createElement("div");
            div.style.background = "#000";
            boardEl.appendChild(div);
        }
    }
}

/* MOVE ALL TOKENS TO THEIR POSITIONS */
function updateAllTokens() {
    for (let i = 0; i < playerCount; i++) {
        movePlayerToken(i);
    }
}

/* MOVE ONE TOKEN */
function movePlayerToken(playerIndex) {
    let p = players[playerIndex];
    let boardIndex = p.pos;
    let spaceEl = boardEl.children[boardIndex];

    let rect = spaceEl.getBoundingClientRect();
    let boardRect = boardEl.getBoundingClientRect();

    let x = rect.left - boardRect.left + rect.width / 2;
    let y = rect.top - boardRect.top + rect.height / 2;

    /* OFFSET TOKENS SO THEY DON'T STACK */
    let offsetAngle = (playerIndex / playerCount) * 2 * Math.PI;
    let offsetRadius = 10;
    x += Math.cos(offsetAngle) * offsetRadius;
    y += Math.sin(offsetAngle) * offsetRadius;

    p.tokenEl.style.left = (boardRect.left + x) + "px";
    p.tokenEl.style.top = (boardRect.top + y) + "px";
}

/* UPDATE LEFT PANEL STATS + MINI AVATAR HIGHLIGHT */
function updateCurrentPlayerUI() {
    let p = players[currentPlayerIndex];
    document.getElementById("currentPlayerLabel").innerText =
        "Current Player: " + p.name;
    document.getElementById("money").innerText = p.money;
    document.getElementById("happy").innerText = p.happiness;
    document.getElementById("job").innerText = p.job;
    document.getElementById("car").innerText = p.car;
    document.getElementById("house").innerText = p.house;

    players.forEach(pl => pl.miniEl.classList.remove("active"));
    p.miniEl.classList.add("active");
}

/* ROLL DICE FOR CURRENT PLAYER */
function rollDice() {
    let p = players[currentPlayerIndex];

    let roll =
        Math.floor(Math.random() * 6 + 1) +
        Math.floor(Math.random() * 6 + 1);

    p.pos = (p.pos + roll) % boardSpaces.length;

    /* RESHUFFLE BOARD EVERY TURN */
    buildBoard();

    /* MOVE TOKENS */
    updateAllTokens();

    /* TRIGGER SPACE */
    triggerSpace();
}

/* TRIGGER SPACE FOR CURRENT PLAYER */
function triggerSpace() {
    let p = players[currentPlayerIndex];
    let cat = boardSpaces[p.pos];
    let btn = document.getElementById("drawButton");
    btn.style.display = "block";
    btn.innerText = p.name + " — Draw " + cat + " Card";
    btn.style.background = categoryColors[cat];
    btn.onclick = () => drawCard(cat);
}

/* DRAW CARD FOR CURRENT PLAYER */
function drawCard(cat) {
    let deck = decks[cat];
    let card = deck[Math.floor(Math.random() * deck.length)];
    applyCardEffect(card);
    showCard(card);
}

/* APPLY CARD EFFECT (money only, from + / - value) */
function applyCardEffect(card) {
    let p = players[currentPlayerIndex];
    let val = parseInt(card.effect, 10);
    if (!isNaN(val)) {
        p.money += val;
        if (p.money < 0) p.money = 0;
    }
    updateCurrentPlayerUI();
}

/* SHOW CARD POPUP */
function showCard(card) {
    let popup = document.getElementById("cardPopup");
    popup.style.borderColor = categoryColors[card.category];

    popup.innerHTML = `
        <h3 style="color:${categoryColors[card.category]}">
            ${card.category} — ${card.type}
        </h3>
        <hr style="border-top:2px solid ${categoryColors[card.category]}">
        <h2>${card.title}</h2>
        <p>${card.effect}</p>
        <button class="drawBtn" onclick="endTurn()">End Turn</button>
    `;
    popup.style.display = "block";
}

/* CLOSE CARD + NEXT PLAYER */
function endTurn() {
    document.getElementById("cardPopup").style.display = "none";

    /* NEXT PLAYER */
    currentPlayerIndex = (currentPlayerIndex + 1) % playerCount;
    updateCurrentPlayerUI();

    /* HIDE DRAW BUTTON UNTIL NEXT ROLL */
    document.getElementById("drawButton").style.display = "none";
}

/* INIT GAME */
buildBoard();
initPlayers();
updateAllTokens();
updateCurrentPlayerUI();
</script>
